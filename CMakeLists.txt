cmake_minimum_required(VERSION 3.20)
project(posmv_sweep VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(POSMV_BUILD_GUI "Build GTK4 operator interface" ON)
include(CTest)
add_library(posmv_core src/protocol/tss1.cpp src/environment/virtual_com.cpp)
target_include_directories(posmv_core PUBLIC src)
add_executable(posmv-sweep src/main.cpp)
target_link_libraries(posmv-sweep PRIVATE posmv_core)
if(NOT WIN32)
  message(FATAL_ERROR "Linux backend is reserved in src/platform/linux but not implemented; application builds currently require Windows")
endif()
add_library(posmv_clock src/platform/windows/precise_wait.cpp)
target_include_directories(posmv_clock PUBLIC src)
target_link_libraries(posmv_core PUBLIC posmv_clock)
add_library(posmv_serial src/platform/windows/serial_transport.cpp src/platform/windows/port_list.cpp)
add_library(posmv_platform src/platform/windows/environment_manager.cpp src/platform/windows/virtual_com_worker.cpp)
target_include_directories(posmv_platform PUBLIC src)
target_link_libraries(posmv_platform PRIVATE posmv_core posmv_serial shell32 bcrypt advapi32 setupapi)
target_include_directories(posmv_serial PUBLIC src)
target_link_libraries(posmv-sweep PRIVATE posmv_serial)
add_executable(posmv-simulator src/simulator/main.cpp)
target_link_libraries(posmv-simulator PRIVATE posmv_core posmv_serial)
target_link_libraries(posmv-simulator PRIVATE posmv_platform)
if(BUILD_TESTING)
  add_executable(test_fixed_frame tests/test_fixed_frame.cpp)
  target_link_libraries(test_fixed_frame PRIVATE posmv_core)
  add_test(NAME fixed_frame COMMAND test_fixed_frame)
  add_executable(test_replay_source tests/test_replay_source.cpp)
  target_link_libraries(test_replay_source PRIVATE posmv_core)
  add_test(NAME replay_source COMMAND test_replay_source)
  add_executable(test_proxy tests/test_proxy.cpp)
  target_link_libraries(test_proxy PRIVATE posmv_core posmv_serial)
  add_test(NAME proxy COMMAND test_proxy)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE posmv_core)
  add_test(NAME core COMMAND test_core)
  add_executable(test_virtual_com tests/test_virtual_com.cpp)
  target_link_libraries(test_virtual_com PRIVATE posmv_core)
  add_test(NAME virtual_com COMMAND test_virtual_com)
  add_executable(test_timing_stats tests/test_timing_stats.cpp)
  target_link_libraries(test_timing_stats PRIVATE posmv_core)
  add_test(NAME timing_stats COMMAND test_timing_stats)
  add_executable(test_sweep tests/test_sweep.cpp)
  target_link_libraries(test_sweep PRIVATE posmv_core)
  add_test(NAME sweep COMMAND test_sweep)
  add_executable(test_signal tests/test_signal.cpp)
  target_link_libraries(test_signal PRIVATE posmv_core)
  add_test(NAME signal COMMAND test_signal)
  add_executable(test_capture tests/test_capture.cpp)
  target_link_libraries(test_capture PRIVATE posmv_core)
  add_test(NAME capture COMMAND test_capture "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/20260904_TSS1 POSMV.txt")
  add_test(NAME replay COMMAND ${CMAKE_COMMAND}
    -DAPP=$<TARGET_FILE:posmv-sweep> -DFIXTURE=$<TARGET_FILE:test_core> -DBINARY=${CMAKE_CURRENT_BINARY_DIR}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.cmake)
  add_test(NAME simulator COMMAND ${CMAKE_COMMAND}
    -DSIM=$<TARGET_FILE:posmv-simulator> -DAPP=$<TARGET_FILE:posmv-sweep>
    -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/simulator.cmake)
endif()
if(POSMV_BUILD_GUI)
  if(WIN32)
    set_target_properties(posmv-sweep posmv-simulator PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
  target_sources(posmv-sweep PRIVATE src/gui/main_window.cpp)
  target_link_libraries(posmv-sweep PRIVATE PkgConfig::GTK4)
  target_compile_definitions(posmv-sweep PRIVATE POSMV_HAS_GUI)
  target_sources(posmv-simulator PRIVATE src/simulator/window.cpp)
  target_link_libraries(posmv-simulator PRIVATE PkgConfig::GTK4)
  target_compile_definitions(posmv-simulator PRIVATE POSMV_HAS_GUI)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(posmv-sweep PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32 AND MINGW)
  get_filename_component(POSMV_RUNTIME_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(app IN ITEMS posmv-sweep posmv-simulator)
    add_custom_command(TARGET ${app} POST_BUILD
      COMMAND ${CMAKE_COMMAND}
        "-DAPP=$<TARGET_FILE:${app}>"
        "-DDESTINATION=$<TARGET_FILE_DIR:${app}>"
        "-DRUNTIME_DIR=${POSMV_RUNTIME_DIR}"
        "-DOBJDUMP=${CMAKE_OBJDUMP}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/DeployRuntime.cmake"
      COMMENT "Deploying ${app} runtime DLL dependencies" VERBATIM)
  endforeach()
endif()
if(WIN32)
  target_link_libraries(posmv-simulator PRIVATE shell32)
endif()
